                    <summary>

                    </summary>
                    <div class="install-content">
                        <div class="install-step">
                            <span class="step-number">1</span>
                            <div class="step-content">
                                <h4>Install Effects</h4>
                                <p>After downloading, extract the <code>null-jsfx</code> folder to your Reaper Effects directory:</p>
                                <dl class="install-paths">
                                    <div class="install-os">
                                        <dt>Windows</dt>
                                        <dd><code>C:/Users/[Username]/AppData/Roaming/REAPER/Effects/</code></dd>
                                    </div>
                                    <div class="install-os">
                                        <dt>macOS</dt>
                                        <dd><code>~/Library/Application Support/REAPER/Effects/</code></dd>
                                    </div>
                                    <div class="install-os">
                                        <dt>Linux</dt>
                                        <dd><code>~/.config/REAPER/Effects/</code></dd>
                                    </div>
                                </dl>
                                <p class="install-note">Restart Reaper or re-scan your plugins folders. <strong>NULL JSFX</strong> will be available under the <strong>JS</strong> category in the FX browser.</p>
                            </div>
                        </div>
                        <div class="install-step">
                            <span class="step-number">2</span>
                            <div class="step-content">
                                <h4>Load Plugins</h4>
                                <ol class="usage-steps">
                                    <li>Click the <strong>FX</strong> button on any track</li>
                                    <li>Open the <strong>FX Browser</strong> (if not visible)</li>
                                    <li>Navigate to the <strong>JS</strong> category</li>
                                    <li>Find plugins under <strong>JS: NULL JSFX</strong> folder</li>
                                    <li>Select any plugin to load it</li>
                                </ol>
                            </div>
                        </div>
                        <div class="install-step">
                            <span class="step-number">3</span>
                            <div class="step-content">
                                <h4>FX Chains (Optional)</h4>
                                <p>Copy <code>.RfxChain</code> files to your <strong>FXChains</strong> folder:</p>
                                <ol class="usage-steps">
                                    <li>Windows: <code>C:/Users/[Username]/AppData/Roaming/REAPER/FXChains/</code></li>
                                    <li>macOS: <code>~/Library/Application Support/REAPER/FXChains/</code></li>
                                    <li>In Reaper's FX Window, go to <strong>FX Chains</strong> category</li>
                                    <li>Double-click any chain to load all plugins automatically</li>
                                </ol>
                                <div class="install-warning">
                                    <strong>&#9888; Important:</strong> Plugins must be installed in <code>Effects/null-jsfx/</code> subfolder for chains to work properly.
                                </div>
                            </div>
                        </div>
                        <div class="install-tips">
                            <div class="install-tip">
                                <strong>&#128161; Tip:</strong> To see all plugins organized in the FX explorer, filter by searching <code>"NULL"</code> or create a Smart Folder: Menu "View" &#8594; FX Browser &#8594; right-click on "Folders" &#8594; "Add folder" &#8594; "Smart folder" &#8594; Filter contains <code>"NULL"</code>. They will auto-categorize!
                            </div>
                            <div class="install-tip">
                                <strong>&#127908; Shortcut:</strong> Press <kbd>/</kbd> to quickly focus the plugin search bar. <kbd>ESC</kbd> closes all modals.
                            </div>
                        </div>
                    </div>

        function searchPlugins(query) {
            const cards = document.querySelectorAll('.plugin-card');
            let visibleCount = 0;

            cards.forEach(card => {
                const dataName = card.getAttribute('data-name').toLowerCase();
                const pluginTitle = card.querySelector('h3').textContent.toLowerCase();
                const pluginDesc = card.querySelector('.plugin-desc').textContent.toLowerCase();

                if (dataName.includes(query) || pluginTitle.includes(query) || pluginDesc.includes(query)) {
                    card.style.display = '';
                    visibleCount++;
                } else {
                    card.style.display = 'none';
                }
            });

            const countEl = document.getElementById('search-count');
            countEl.textContent = visibleCount + ' plugin' + (visibleCount !== 1 ? 's' : '') + ' found';
            countEl.style.opacity = query.length > 0 ? '1' : '0';
        }

            document.getElementById('plugin-search').value = '';
            searchPlugins('');
            document.querySelector('.search-clear').style.display = 'none';
        }

        document.getElementById('plugin-search').addEventListener('input', function(e) {
            searchPlugins(e.target.value.toLowerCase());
            document.querySelector('.search-clear').style.display = e.target.value.length > 0 ? 'block' : 'none';
        });

        // Back to top
        window.addEventListener('scroll', function() {
            const backToTop = document.getElementById('back-to-top');
            backToTop.style.opacity = window.scrollY > 300 ? '1' : '0';
            backToTop.style.pointerEvents = window.scrollY > 300 ? 'auto' : 'none';
        });

        function scrollToTop() {
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        // Tab switching
        window.switchTab = function(tabName) {
            document.querySelectorAll('.tab').forEach(function(t) {
                t.classList.remove('active');
            });
            document.querySelectorAll('.tab-panel').forEach(function(p) {
                p.classList.remove('active');
            });
            document.getElementById('tab-btn-' + tabName).classList.add('active');
            document.getElementById('panel-' + tabName).classList.add('active');
        };
