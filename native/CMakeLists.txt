# SPDX-License-Identifier: MIT
#
# Every folder in plugins/ is one plugin: DistrhoPluginInfo.h, Plugin.cpp, UI.cpp and any
# headers it needs. It builds as null_jsfx_<folder>.vst3/.clap. -DNULL_PLUGINS="tilt;louder"
# limits the build to some plugins.

cmake_minimum_required(VERSION 3.15...3.31)

# Universal macOS binaries unless the caller chose otherwise (must precede project()).
if(APPLE AND NOT DEFINED CMAKE_OSX_ARCHITECTURES)
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "macOS architectures")
endif()
if(APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "macOS deployment target")
endif()

project(null_jsfx_native VERSION 2.6.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
# No fused multiply-add: the plugins must round like the JSFX (EEL2 never fuses). Apple Clang
# fuses by default on arm64, which moved e.g. Grit's sample-hold step by one ulp.
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-ffp-contract=off)
endif()

option(NULL_UI    "Build the NanoVG GUI (needs X11/OpenGL dev headers on Linux)" ON)
option(NULL_VST3  "Build VST3" ON)
option(NULL_CLAP  "Build CLAP" ON)
option(NULL_JACK  "Build standalones (JACK / native audio), handy for UI screenshots" OFF)
option(NULL_TESTS "Build the JSFX-vs-native comparison tests (needs a built ysfx)" OFF)
set(NULL_PLUGINS "" CACHE STRING "Plugins to build (folder names in plugins/); empty = all")
set(NULL_MODULES "" CACHE STRING "Rack modules to include (file names in plugins/rack/modules/ without .hpp); empty = all")

# ---------------------------------------------------------------------------------------------------------------------
# DPF, pinned. Override with -DFETCHCONTENT_SOURCE_DIR_DPF=/path/to/DPF for offline builds.

set(DPF_GIT_SHA 4238e1c7f0351bbe488d79f0899c540543ac7583) # DPF main, 2025-10-23 "Merge branch 'develop'"

include(FetchContent)

if(MSVC)
  # MSVC ships only OpenGL 1.1 headers; NanoVG needs the GL 2.0 types from Khronos (MIT)
  include_directories(BEFORE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/khronos")
endif()
# Every third_party/dpf/*.patch is applied on top of the pinned commit, in name order (a local --dpf
# checkout needs them applied by hand). The reset first makes the step idempotent.
#   state-load.patch  CLAP/VST3 state load: a string cut by a short stream read is not ended early
file(GLOB _dpf_patches "${CMAKE_CURRENT_SOURCE_DIR}/third_party/dpf/*.patch")
list(SORT _dpf_patches)
set(_dpf_patch_command PATCH_COMMAND git reset -q --hard ${DPF_GIT_SHA})
foreach(_dpf_patch IN LISTS _dpf_patches)
  list(APPEND _dpf_patch_command COMMAND git apply "${_dpf_patch}")
endforeach()
FetchContent_Declare(dpf
  GIT_REPOSITORY https://github.com/DISTRHO/DPF.git
  GIT_TAG        ${DPF_GIT_SHA}
  GIT_SUBMODULES "dgl/src/pugl-upstream"
  GIT_SHALLOW    FALSE
  GIT_CONFIG     core.autocrlf=false
  ${_dpf_patch_command})
set(DPF_LIBRARIES OFF CACHE BOOL "" FORCE)
set(DPF_EXAMPLES  OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(dpf)

# ---------------------------------------------------------------------------------------------------------------------
# Noto Sans (SIL OFL 1.1, common/fonts/OFL.txt), embedded in every interface so the text looks the same on every OS
# and matches the GFX JSFX, which ask REAPER for "Noto Sans"

set(_fonts_h "${CMAKE_CURRENT_BINARY_DIR}/generated/NullFonts.h")
if(NOT EXISTS "${_fonts_h}")
  set(_out "// generated by CMakeLists.txt from common/fonts (Noto Sans, SIL OFL 1.1)\n#pragma once\n")
  foreach(_w Regular Bold)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/common/fonts/NotoSans-${_w}.ttf" _hex HEX)
    string(LENGTH "${_hex}" _len)
    math(EXPR _bytes "${_len} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _hex "${_hex}")
    string(APPEND _out "static const unsigned char kNotoSans${_w}[${_bytes}] = {${_hex}};\n")
  endforeach()
  file(WRITE "${_fonts_h}" "${_out}")
endif()
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")

# ---------------------------------------------------------------------------------------------------------------------
# Plugins

set(_targets)
foreach(_t JACK VST3 CLAP)
  if(NULL_${_t})
    string(TOLOWER ${_t} _lt)
    list(APPEND _targets ${_lt})
  endif()
endforeach()

if(NULL_PLUGINS)
  set(_plugins ${NULL_PLUGINS})
else()
  file(GLOB _dirs LIST_DIRECTORIES true RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/plugins" "${CMAKE_CURRENT_SOURCE_DIR}/plugins/*")
  set(_plugins)
  foreach(_d ${_dirs})
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/plugins/${_d}/Plugin.cpp")
      list(APPEND _plugins ${_d})
    endif()
  endforeach()
  list(SORT _plugins)
endif()

# The Rack's module list: every plugins/rack/modules/<id>.hpp, in file-name order, as
# generated/NullModules.hpp (kModules[], kNumModules); -DNULL_MODULES="tilt;delay" limits it
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/plugins/rack/modules")
  if(NULL_MODULES)
    set(_mods)
    foreach(_m ${NULL_MODULES})
      list(APPEND _mods "${_m}.hpp")
    endforeach()
  else()
    file(GLOB _mods RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/plugins/rack/modules" "${CMAKE_CURRENT_SOURCE_DIR}/plugins/rack/modules/*.hpp")
  endif()
  list(SORT _mods)
  set(NULL_RACK_MODULES)
  set(_inc "")
  set(_list "")
  foreach(_m ${_mods})
    string(REGEX REPLACE "\\.hpp$" "" _id "${_m}")
    string(APPEND _inc "#include \"${CMAKE_CURRENT_SOURCE_DIR}/plugins/rack/modules/${_m}\"\n")
    string(APPEND _list "    &nullrack::${_id}::kDesc,\n")
    list(APPEND NULL_RACK_MODULES ${_id})
  endforeach()
  set(_out "// generated by CMakeLists.txt from plugins/rack/modules\n#pragma once\n\n${_inc}\nnamespace nullrack {\n\nstatic const nulljsfx::ModuleDesc* const kModules[] = {\n${_list}};\nstatic constexpr uint32_t kNumModules = sizeof(kModules) / sizeof(kModules[0]);\n\n} // namespace nullrack\n")
  set(_mods_h "${CMAKE_CURRENT_BINARY_DIR}/generated/NullModules.hpp")
  if(EXISTS "${_mods_h}")
    file(READ "${_mods_h}" _old)
  else()
    set(_old "")
  endif()
  if(NOT _old STREQUAL _out)
    file(WRITE "${_mods_h}" "${_out}")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/plugins/rack/modules")
endif()

set(NULL_BUILT_PLUGINS "")
foreach(_p ${_plugins})
  set(_dir "${CMAKE_CURRENT_SOURCE_DIR}/plugins/${_p}")
  set(_ui)
  if(NULL_UI AND EXISTS "${_dir}/UI.cpp")
    set(_ui "${_dir}/UI.cpp")
  endif()
  dpf_add_plugin(null_jsfx_${_p}
    TARGETS ${_targets}
    UI_TYPE opengl
    FILES_DSP "${_dir}/Plugin.cpp"
    FILES_UI  ${_ui})
  target_include_directories(null_jsfx_${_p} PUBLIC "${_dir}" "${CMAKE_CURRENT_SOURCE_DIR}/common")
  if(NOT _ui)
    target_compile_definitions(null_jsfx_${_p} PUBLIC NULL_NO_UI=1)
  endif()
  list(APPEND NULL_BUILT_PLUGINS ${_p})
endforeach()
